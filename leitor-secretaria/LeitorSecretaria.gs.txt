var LEITOR = {
  CAIXA: "SECRETARIA",
  ENDERECO_CAIXA: "secretaria@sindeducacao.com",
  CONTA_SISGEP: "financeirosindecucacao@gmail.com",
  ABA_ENTRADA: "Entrada",
  ABA_SAIDA: "Saida",
  ABA_VISTOS: "Vistos",
  DIAS: 15,
  MAX_CONVERSAS: 100,
  MAX_ENVIOS_POR_CICLO: 20,
  ANEXO_MAX_BYTES: 10 * 1024 * 1024,
  CORPO_MAX: 6000,
  LIMITE_MS: 240000
};

var LEITOR_COLUNAS = {
  Entrada: ["THREAD_ID", "ULTIMA_MENSAGEM_ID", "CAIXA", "ATUALIZADO_EM", "MENSAGENS_JSON", "ANEXOS_JSON"],
  Saida: ["ID", "THREAD_ID", "MENSAGEM_ID", "HTML", "ANEXOS_FILE_IDS", "STATUS", "CRIADO_EM",
          "CRIADO_POR", "PROCESSADO_EM", "DETALHE"],
  Vistos: ["THREAD_ID", "ULTIMA_MENSAGEM_ID", "VISTO_EM"]
};

/**
 * Rodar UMA vez, pelo editor, logado na conta da Secretaria.
 * Cria a planilha-ponte e a pasta de anexos, dá acesso à conta do SISGEP
 * e liga o gatilho de 15 minutos. Devolve o ID que vai no SISGEP.
 */
function configurarLeitor() {
  var props = PropertiesService.getScriptProperties();
  var id = props.getProperty("PONTE_ID");
  var ss;
  if (id) {
    ss = SpreadsheetApp.openById(id);
  } else {
    ss = SpreadsheetApp.create("SISGEP — Ponte da Secretaria (pedidos de voucher)");
    props.setProperty("PONTE_ID", ss.getId());
  }
  Object.keys(LEITOR_COLUNAS).forEach(function (nome) { leitorAba_(ss, nome); });

  var pastaId = props.getProperty("PASTA_ID");
  var pasta = pastaId ? DriveApp.getFolderById(pastaId)
                      : DriveApp.createFolder("SISGEP — Anexos de pedidos de voucher");
  props.setProperty("PASTA_ID", pasta.getId());

  /* O SISGEP precisa LER a entrada e os anexos e ESCREVER na saída. */
  DriveApp.getFileById(ss.getId()).addEditor(LEITOR.CONTA_SISGEP);
  pasta.addViewer(LEITOR.CONTA_SISGEP);

  var existe = ScriptApp.getProjectTriggers().some(function (t) {
    return t.getHandlerFunction() === "leitorCiclo";
  });
  if (!existe) ScriptApp.newTrigger("leitorCiclo").timeBased().everyMinutes(15).create();

  var msg = "Leitor configurado. No SISGEP, em Propriedades do script, crie " +
    "VOUCHER_PEDIDOS_PONTE_ID = " + ss.getId();
  Logger.log(msg);
  return msg;
}

function leitorAba_(ss, nome) {
  var sh = ss.getSheetByName(nome) || ss.insertSheet(nome);
  if (sh.getLastRow() < 1) {
    sh.getRange(1, 1, 1, LEITOR_COLUNAS[nome].length).setValues([LEITOR_COLUNAS[nome]]);
  }
  return sh;
}

function leitorPonte_() {
  var id = PropertiesService.getScriptProperties().getProperty("PONTE_ID");
  if (!id) throw new Error("Leitor não configurado: rode configurarLeitor() uma vez.");
  return SpreadsheetApp.openById(id);
}

function leitorLinhas_(sh) {
  if (sh.getLastRow() < 2) return [];
  var tudo = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getValues();
  var cab = tudo[0].map(function (c) { return String(c || "").trim(); });
  var out = [];
  for (var l = 1; l < tudo.length; l++) {
    var o = { __linha: l + 1 };
    cab.forEach(function (n, i) { if (n) o[n] = tudo[l][i]; });
    out.push(o);
  }
  return out;
}

function leitorGravar_(sh, reg) {
  var cab = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]
    .map(function (c) { return String(c || "").trim(); });
  var linha = reg.__linha || (sh.getLastRow() + 1);
  sh.getRange(linha, 1, 1, cab.length).setValues([cab.map(function (c) {
    var v = reg[c]; return v === undefined || v === null ? "" : v;
  })]);
  reg.__linha = linha;
  return reg;
}

/* ═══════════════════════════════════════════════════════════════════════
   O CICLO — gatilho de 15 minutos
═══════════════════════════════════════════════════════════════════════ */

function leitorCiclo() {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(5000)) return { ocupado: true };
  try {
    var r = { lidas: leitorLerCaixa_(), enviadas: leitorEnviarRespostas_() };
    Logger.log(JSON.stringify(r));
    return r;
  } finally {
    lock.releaseLock();
  }
}

function leitorEmailDe_(s) {
  var m = String(s || "").match(/[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}/);
  return m ? m[0].toLowerCase() : "";
}

function leitorNossos_() {
  var n = [LEITOR.ENDERECO_CAIXA];
  try { n.push(String(Session.getEffectiveUser().getEmail() || "").toLowerCase()); } catch (e) {}
  try { GmailApp.getAliases().forEach(function (a) { n.push(String(a).toLowerCase()); }); } catch (e) {}
  return n.filter(String);
}

/** Tira a citação da mensagem anterior: "obrigado" cita o nosso e-mail. */
function leitorSemCitacao_(texto) {
  var linhas = String(texto || "").split(/\r?\n/), out = [];
  for (var i = 0; i < linhas.length; i++) {
    if (/^\s*(Em .+ escreveu:|On .+ wrote:|-{2,}\s*Mensagem original|De:\s.+@)/i.test(linhas[i])) break;
    if (/^\s*>/.test(linhas[i])) continue;
    out.push(linhas[i]);
  }
  return out.join("\n").trim();
}

/* A MESMA peneira do SISGEP (vpePareceVoucher_). Larga de propósito: o que
   passa aqui ainda é conferido pela IA e por gente; o que NÃO passa nunca
   chega a ninguém. Errar para o lado de copiar a mais é o erro barato. */
function leitorPareceVoucher_(texto) {
  return /voucher|bolsa|certificado|desconto|faculdade|universidade|gradua|p[óo]s[\s-]?gradua|semestre|matr[íi]cula|mensalidade escolar|curso\b/i
    .test(String(texto || ""));
}

function leitorLerCaixa_() {
  var inicio = Date.now();
  var ss = leitorPonte_();
  var shE = leitorAba_(ss, LEITOR.ABA_ENTRADA);
  var shV = leitorAba_(ss, LEITOR.ABA_VISTOS);
  var entrada = {}, vistos = {};
  leitorLinhas_(shE).forEach(function (x) { entrada[String(x.THREAD_ID)] = x; });
  leitorLinhas_(shV).forEach(function (x) { vistos[String(x.THREAD_ID)] = x; });
  var nossos = leitorNossos_();
  var pasta = DriveApp.getFolderById(PropertiesService.getScriptProperties().getProperty("PASTA_ID"));

  var consulta = "in:inbox newer_than:" + LEITOR.DIAS + "d " +
    "{to:" + LEITOR.ENDERECO_CAIXA + " cc:" + LEITOR.ENDERECO_CAIXA + "} " +
    "-from:mailer-daemon -from:postmaster";
  var threads = GmailApp.search(consulta, 0, LEITOR.MAX_CONVERSAS);
  var copiadas = 0;

  for (var i = 0; i < threads.length; i++) {
    if (Date.now() - inicio > LEITOR.LIMITE_MS) break;
    var th = threads[i], tid = String(th.getId());
    var entrantes = th.getMessages().filter(function (m) {
      return nossos.indexOf(leitorEmailDe_(m.getFrom())) === -1;
    });
    if (!entrantes.length) continue;
    var ultima = String(entrantes[entrantes.length - 1].getId());
    var regE = entrada[tid], regV = vistos[tid];
    if (regE && String(regE.ULTIMA_MENSAGEM_ID) === ultima) continue;
    if (!regE && regV && String(regV.ULTIMA_MENSAGEM_ID) === ultima) continue;

    if (!regE) {
      var texto = entrantes.map(function (m) {
        return (m.getSubject() || "") + "\n" + leitorSemCitacao_(m.getPlainBody());
      }).join("\n");
      if (!leitorPareceVoucher_(texto)) {
        leitorGravar_(shV, { __linha: regV && regV.__linha, THREAD_ID: tid,
          ULTIMA_MENSAGEM_ID: ultima, VISTO_EM: new Date() });
        continue;
      }
    }
    entrada[tid] = leitorGravar_(shE, leitorCopiarConversa_(th, entrantes, regE, pasta));
    copiadas++;
  }
  return copiadas;
}

/** Copia a conversa inteira (só o que veio de fora) e os anexos novos. */
function leitorCopiarConversa_(th, entrantes, regE, pasta) {
  var jaSalvos = {};
  (leitorJson_(regE && regE.ANEXOS_JSON) || []).forEach(function (a) { jaSalvos[a.msgId + "#" + a.pos] = a; });
  var anexos = [];
  var mensagens = entrantes.map(function (m) {
    var mid = String(m.getId());
    (m.getAttachments({ includeInlineImages: false, includeAttachments: true }) || []).forEach(function (a, pos) {
      var chave = mid + "#" + pos;
      if (jaSalvos[chave]) { anexos.push(jaSalvos[chave]); return; }
      var tam = a.getSize ? a.getSize() : a.getBytes().length;
      var info = { msgId: mid, pos: pos, nome: a.getName() || ("anexo-" + (pos + 1)),
                   tipo: String(a.getContentType() || ""), tamanho: tam, fileId: "" };
      if (tam <= LEITOR.ANEXO_MAX_BYTES) {
        var f = pasta.createFile(a.copyBlob().setName(String(th.getId()) + "_" + mid + "_" + pos + "_" + info.nome));
        info.fileId = f.getId();
      }
      anexos.push(info);
    });
    return {
      id: mid,
      de: m.getFrom() || "",
      replyTo: (m.getReplyTo && m.getReplyTo()) || "",
      para: m.getTo() || "",
      cc: (m.getCc && m.getCc()) || "",
      assunto: m.getSubject() || "",
      data: new Date(m.getDate()).toISOString(),
      corpo: leitorSemCitacao_(m.getPlainBody()).substring(0, LEITOR.CORPO_MAX)
    };
  });
  return {
    __linha: regE && regE.__linha,
    THREAD_ID: String(th.getId()),
    ULTIMA_MENSAGEM_ID: mensagens[mensagens.length - 1].id,
    CAIXA: LEITOR.CAIXA,
    ATUALIZADO_EM: new Date(),
    MENSAGENS_JSON: JSON.stringify(mensagens).substring(0, 45000),
    ANEXOS_JSON: JSON.stringify(anexos)
  };
}

function leitorJson_(t) { try { return t ? JSON.parse(String(t)) : null; } catch (e) { return null; } }

/* ═══════════════════════════════════════════════════════════════════════
   O ENVIO — as respostas que o SISGEP deixou na fila
═══════════════════════════════════════════════════════════════════════ */

function leitorEnviarRespostas_() {
  var ss = leitorPonte_();
  var shS = leitorAba_(ss, LEITOR.ABA_SAIDA);
  var conhecidas = {};
  leitorLinhas_(leitorAba_(ss, LEITOR.ABA_ENTRADA)).forEach(function (x) { conhecidas[String(x.THREAD_ID)] = true; });

  var enviadas = 0;
  var pendentes = leitorLinhas_(shS).filter(function (r) { return String(r.STATUS) === "PENDENTE"; });
  pendentes.slice(0, LEITOR.MAX_ENVIOS_POR_CICLO).forEach(function (r) {
    r.PROCESSADO_EM = new Date();
    try {
      var tid = String(r.THREAD_ID || "");
      if (!conhecidas[tid]) {
        r.STATUS = "RECUSADO";
        r.DETALHE = "A conversa não é um pedido copiado por este leitor.";
        return leitorGravar_(shS, r);
      }
      var msg = GmailApp.getMessageById(String(r.MENSAGEM_ID || ""));
      if (!msg || String(msg.getThread().getId()) !== tid) {
        r.STATUS = "RECUSADO";
        r.DETALHE = "A mensagem não pertence à conversa do pedido.";
        return leitorGravar_(shS, r);
      }
      var anexos = String(r.ANEXOS_FILE_IDS || "").split(",").map(function (s) { return s.trim(); })
        .filter(String).map(function (id) {
          var f = DriveApp.getFileById(id);
          if (String(f.getMimeType()) !== "application/pdf") throw new Error("Só se anexa PDF (" + f.getName() + ").");
          return f.getBlob();
        });
      var op = { htmlBody: String(r.HTML || ""), name: "SindEducação-ES", replyTo: LEITOR.ENDERECO_CAIXA };
      /* `from` só quando a secretaria@ é alias verificado — forçar alias que
         não existe faz o Gmail recusar (lição dos ofícios, 02/09/2026). */
      try {
        var aliases = GmailApp.getAliases().map(function (a) { return String(a).toLowerCase(); });
        if (aliases.indexOf(LEITOR.ENDERECO_CAIXA) > -1) op.from = LEITOR.ENDERECO_CAIXA;
        else r.DETALHE = "Saiu sem o alias secretaria@ (configure 'Enviar e-mail como' neste Gmail).";
      } catch (e) {}
      if (anexos.length) op.attachments = anexos;
      msg.reply("", op);
      r.STATUS = "ENVIADO";
      enviadas++;
    } catch (e) {
      r.STATUS = "ERRO";
      r.DETALHE = String(e.message || e).substring(0, 500);
    }
    leitorGravar_(shS, r);
  });
  return enviadas;
}
